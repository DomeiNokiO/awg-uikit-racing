Integrasi Blade Laravel

Cara menghubungkan layout AWG-UIKIT-RACING ke resources/views/layouts/awg.blade.php.

Langkah integrasi
  1. Copy folder css/, js/, fonts/, assets/, libs/ ke public/assets/awg/
  2. Simpan templates/blade/awg-layout.blade.php sebagai resources/views/layouts/awg.blade.php
  3. Di controller kirim array $menu sesuai hak akses user
  4. Halaman content: @extends('layouts.awg') @section('content') … @endsection
<?php
// routes/web.php
Route::get('/dashboard', [DashboardController::class,'index'])->name('dashboard');

// DashboardController.php
public function index() {
    return view('dashboard', [
        'menu' => [
            'NMS' => [
                ['label'=>'Dashboard','route'=>'dashboard','icon'=>'ic-gauge','active'=>true],
                ['label'=>'Node','route'=>'nodes.index','icon'=>'ic-server'],
            ],
        ],
    ]);
}
